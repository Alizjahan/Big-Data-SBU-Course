	\section{سؤال اول: محاسبه حاصل‌ضرب ماتریس‌ها با الگوریتم \lr{MapReduce}}
	
	دو ماتریس غیرمربعی $A$ از مرتبه $2 \times 3$ و $B$ از مرتبه $3 \times 2$ به شرح زیر مفروض هستند:
	\begin{equation}
		A = \begin{bmatrix} 1 & 2 & 0 \\ 3 & 1 & 2 \end{bmatrix}, \quad
		B = \begin{bmatrix} 2 & 1 \\ 0 & 3 \\ 4 & 2 \end{bmatrix}
	\end{equation}
	هدف، محاسبه ماتریس حاصل‌ضرب $C = A \times B$ از مرتبه $2 \times 2$ با استفاده از مدل برنامه‌نویسی توزیع‌شده \lr{MapReduce} است. درایه‌های ماتریس $A$ به صورت چهارگانه $(A, i, j, \text{value})$ و درایه‌های ماتریس $B$ به صورت $(B, j, k, \text{value})$ ذخیره شده‌اند و اندیس‌ها از ۱ آغاز می‌شوند ($i \in \{1, 2\}$، $j \in \{1, 2, 3\}$ و $k \in \{1, 2\}$).
	
	\subsection{الف) تشریح الگوریتم دوسطحی \lr{MapReduce}}
	فرمول ریاضی محاسبه هر درایه ماتریس حاصل‌ضرب به صورت زیر تعریف می‌شود:
	\begin{equation}
		C_{ik} = \sum_{j=1}^{n} A_{ij} B_{jk}
	\end{equation}
	این رابطه ریاضی متشکل از دو بخش محاسباتی مستقل است:
	\begin{enumerate}
		\item \textbf{تولید حاصل‌ضرب‌های جزئی (\lr{Partial Products}):} به ازای هر سه‌گانه $(i, j, k)$، مقدار $A_{ij} \times B_{jk}$ باید محاسبه شود.
		\item \textbf{تجمیع و جمع‌بندی (\lr{Aggregation}):} جمع‌بندی تمام ضرب‌های جزئی به ازای اندیس مشترک $j$ برای سلول نهایی $C_{ik}$.
	\end{enumerate}
	
	بر همین اساس، الگوریتم در دو مرحله (\lr{Phase}) کامل و مجزا پیاده‌سازی می‌گردد:
	
	\begin{tcolorbox}[primarycard, title={مرحله اول: تولید ضرب‌های جزئی (\lr{Partial Products Generation})}]
		\small
		\begin{itemize}
			\item \textbf{فاز \lr{Map}:} مپِر (\lr{Mapper}) رکوردهای ورودی ماتریس‌ها را به شرح زیر می‌خواند:
			\begin{itemize}
				\item برای هر درایه ماتریس $A(i, j, a)$، کلید خروجی برابر اندیس ستونی $j$ و مقدار برابر زوج $(A, i, a)$ است.
				\item برای هر درایه ماتریس $B(j, k, b)$، کلید خروجی برابر اندیس سطری $j$ و مقدار برابر زوج $(B, k, b)$ است.
			\end{itemize}
			\item \textbf{فاز \lr{Shuffle / Grouping}:} بر اساس کلید میانی $j$، تمامی درایه‌های ستون $j$-ام از $A$ و سطر $j$-ام از $B$ روی یک ردیوسر مشترک متمرکز می‌گردند.
			\item \textbf{فاز \lr{Reduce}:} به ازای کلید $j$، ردیوسر دو لیست مقادیر $A$ و $B$ را دریافت کرده و ضرب دکارتی آن‌ها را تشکیل می‌دهد. برای هر جفت $(A, i, a)$ و $(B, k, b)$، حاصل‌ضرب جزئی $v = a \times b$ محاسبه شده و با کلید جدید مختصاتی $((i, k), v)$ منتشر می‌گردد.
		\end{itemize}
	\end{tcolorbox}
	
	\vspace{0.2cm}
	
	\begin{tcolorbox}[accentcard, title={مرحله دوم: تجمیع مقادیر نهایی (\lr{Aggregation Phase})}]
		\small
		\begin{itemize}
			\item \textbf{فاز \lr{Map}:} یک مپِر همانی (\lr{Identity Mapper}) است که خروجی‌های مرحله اول را بدون تغییر ارسال می‌کند: کلید $(i, k)$ و مقدار $v$.
			\item \textbf{فاز \lr{Shuffle / Grouping}:} تمامی حاصل‌ضرب‌های جزئی با کلید مشترک $(i, k)$ در یک گروه دسته‌بندی می‌شوند.
			\item \textbf{فاز \lr{Reduce}:} ردیوسر مقادیر لیست ورودی $[v_1, v_2, \dots, v_n]$ را جمع زده و درایه نهایی را به صورت $((i, k), C_{ik})$ ساطع می‌نماید.
		\end{itemize}
	\end{tcolorbox}

	\textbf{نقش استراتژیک کلیدها در هدایت جریان داده‌ها:}
	\begin{itemize}
		\item در \textbf{مرحله اول}، انتخاب اندیس میانی $j$ به عنوان کلید، نقش متصل‌کننده ابعاد مشترک ضرب ماتریسی را ایفا می‌کند و تضمین می‌نماید که ستون $j$ از $A$ با سطر $j$ از $B$ روی یک گره محاسباتی قرار گیرند.
		\item در \textbf{مرحله دوم}، انتخاب جفت مختصات هدف $(i, k)$ به عنوان کلید، تمامی ضرایب جزئی یک درایه مشخص از ماتریس نهایی را در یک نقطه تجمیع کرده و امکان جمع موازی مستقل را فراهم می‌سازد.
	\end{itemize}
	
	\subsection{ج) ردیابی گام‌به‌گام برای درایه $C_{12}$ و محاسبه کل ماتریس $C$}
	
	برای محاسبه درایه $C_{12}$ ($i=1, k=2$):
	\begin{itemize}
		\item عناصر سطر اول ماتریس $A$: $(A, 1, 1, 1)، (A, 1, 2, 2)، (A, 1, 3, 0)$
		\item عناصر ستون دوم ماتریس $B$: $(B, 1, 2, 1)، (B, 2, 2, 3)، (B, 3, 2, 2)$
	\end{itemize}
	
	\textbf{ردیابی در مرحله اول (تولید ضرب‌های جزئی):}
	\begin{itemize}
		\item کلید $j=1$: مقدار $(A, 1, 1)$ و $(B, 2, 1)$ ضرب شده $\implies$ خروجی: $((1, 2), 1 \times 1 = 1)$.
		\item کلید $j=2$: مقدار $(A, 1, 2)$ و $(B, 2, 3)$ ضرب شده $\implies$ خروجی: $((1, 2), 2 \times 3 = 6)$.
		\item کلید $j=3$: مقدار $(A, 1, 0)$ و $(B, 2, 2)$ ضرب شده $\implies$ خروجی: $((1, 2), 0 \times 2 = 0)$.
	\end{itemize}
	
	\textbf{ردیابی در مرحله دوم (تجمیع نهایی):}
	\begin{itemize}
		\item ردیوسر متناظر با کلید $(1, 2)$ مقادیر لیست $[1, 6, 0]$ را دریافت کرده و حاصل جمع را محاسبه می‌کند:
		\begin{equation}
			C_{12} = 1 + 6 + 0 = 7
		\end{equation}
	\end{itemize}
	
	\textbf{محاسبه تمامی درایه‌های ماتریس $C$:}
	\begin{align}
		C_{11} &= A_{11}B_{11} + A_{12}B_{21} + A_{13}B_{31} = 1(2) + 2(0) + 0(4) = 2 + 0 + 0 = 2 \\
		C_{12} &= A_{11}B_{12} + A_{12}B_{22} + A_{13}B_{32} = 1(1) + 2(3) + 0(2) = 1 + 6 + 0 = 7 \\
		C_{21} &= A_{21}B_{11} + A_{22}B_{21} + A_{23}B_{31} = 3(2) + 1(0) + 2(4) = 6 + 0 + 8 = 14 \\
		C_{22} &= A_{21}B_{12} + A_{22}B_{22} + A_{23}B_{32} = 3(1) + 1(3) + 2(2) = 3 + 3 + 4 = 10
	\end{align}
	
	بنابراین ماتریس خروجی به شکل زیر حاصل می‌گردد:
	\begin{equation}
		C = \begin{bmatrix} 2 & 7 \\ 14 & 10 \end{bmatrix}
	\end{equation}
	
	\subsection{د) قابلیت‌های موازی‌سازی و مناسب بودن برای داده‌های بزرگ}
	\begin{enumerate}
		\item \textbf{بخش‌های قابل اجرای موازی:}
		فازهای نگاشت (\lr{Map}) هر دو مرحله کاملاً مستقل هستند. همچنین تا سقف $n$ ردیوسر در مرحله اول و تا سقف $m \times p$ ردیوسر در مرحله دوم می‌توانند به طور هم‌زمان و موازی روی کلاسترهای توزیع‌شده اجرا گردند.
		\item \textbf{مزایای روش برای ماتریس‌های بسیار بزرگ:}
		\begin{itemize}
			\item \textbf{مدیریت حافظه (\lr{Out-of-Core}):} هیچ پردازنده‌ای نیازمند بارگذاری کل ماتریس‌های عظیم در رم نیست.
			\item \textbf{بهینگی در ماتریس‌های خلوت (\lr{Sparsity}):} درایه‌های صفر فیلتر شده و حجم ترافیک شبکه به حداقل می‌رسد.
			\item \textbf{مقیاس‌پذیری افقی (\lr{Scalability}):} بار محاسباتی بدون گلوگاه به تسک‌های مستقل شکسته می‌شود.
		\end{itemize}
	\end{enumerate}

	\section{سؤال دوم: پیاده‌سازی عملگرهای رابطه‌ای در \lr{MapReduce}}
	
	جداول رابطه‌ای مفروض عبارتند از:
	\begin{itemize}
		\item جدول $R_1$: شامل $(11, \text{Ali}, \text{CS})، (14, \text{Sara}, \text{Math})، (18, \text{Mina}, \text{CS})$
		\item جدول $R_2$: شامل $(14, \text{Sara}, \text{Math})، (21, \text{Reza}, \text{CS})، (22, \text{Neda}, \text{Physics})$
		\item جدول $R_3$: شامل $(11, \text{Tehran})، (14, \text{Shiraz})، (21, \text{Tabriz})، (25, \text{Isfahan})$
	\end{itemize}
	
	\subsection{الف) طراحی توابع \lr{Map} و \lr{Reduce} برای اجتماع مجموعه‌ای (\lr{Set Union})}
	هدف، محاسبه اجتماع $U = R_1 \cup R_2$ همراه با حذف سطرهای تکراری است:
	
	\begin{itemize}
		\item \textbf{طراحی تابع \lr{Map}:}
		به ازای هر سطر $t = (\text{EmployeeID}, \text{Name}, \text{Department})$ در جداول $R_1$ و $R_2$، کل محتوای سطر به عنوان کلید ساطع می‌شود و مقدار آن یک پرچم نمادین (مانند ۱) است:
		\begin{equation}
			\text{Map}(t) \longrightarrow (t, 1)
		\end{equation}
		\item \textbf{مکانیزم حذف ردیف‌های تکراری:}
		در مرحله \lr{Shuffle}، رکوردهایی با محتوای یکسان به دلیل کلید برابر به یک ردیوسر مشترک هدایت می‌شوند. اگر سطری در هر دو جدول باشد، ورودی ردیوسر به شکل $(t, [1, 1])$ خواهد بود.
		\item \textbf{طراحی تابع \lr{Reduce}:}
		ردیوسر تنها یک نسخه از کلید $t$ را به خروجی ارسال کرده و تکرار مقادیر را نادیده می‌گیرد:
		\begin{equation}

	\begin{equation}
		J(S_a, S_b) = \frac{|S_a \cap S_b|}{|S_a \cup S_b|}, \quad \hat{J}(D_a, D_b) = \frac{1}{K} \sum_{i=1}^{K} \mathbf{1}_{\{h_i(D_a) = h_i(D_b)\}}
	\end{equation}

		\item \textbf{برای زوج $(D_1, D_2)$:}

		\item \textbf{برای زوج $(D_1, D_3)$:}
		اشتراک شامل $\{\text{mn}, \text{nr}, \text{rs}\}$ (طول ۳) و اجتماع شامل ۵ عضو است: $J(D_1, D_3) = \frac{3}{5} = 0.6000$.
		در ماتریس امضاها در هر سه تابع برابری وجود دارد $\implies \hat{J} = \frac{3}{3} = 1.0000$ (خطای مطلق برابر $0.4000$).

	\subsection{د) تحلیل تفاوت تخمین و اثر افزایش تعداد توابع هش}
	\begin{enumerate}
		\item \textbf{علت تفاوت تخمین با مقدار واقعی:}
		تخمین \lr{MinHash} بر مبنای متغیر تصادفی برنولی با امید ریاضی $\mathbb{E}[\hat{J}] = J$ است (برآوردگر نااریب). اما با تعداد اندک توابع هش ($K=3$)، مقادیر ممکن تخمین تنها اعضای مجموعه گسسته $\{0, \frac{1}{3}, \frac{2}{3}, 1\}$ هستند و عدد $0.6$ امکان تحقق مستقیم ندارد.
		\item \textbf{اثر افزایش تعداد جایگشت‌های تصادفی مستقل ($K$):}
		واریانس تخمین‌گر با تعداد توابع هش نسبت معکوس دارد:
		\begin{equation}
			\text{Var}(\hat{J}) = \frac{J(1 - J)}{K} \implies \sigma_{\hat{J}} = \sqrt{\frac{J(1 - J)}{K}} = \mathcal{O}\left(\frac{1}{\sqrt{K}}\right)
		\end{equation}
		با افزایش $K$ (مثلاً به ۱۰۰ یا ۲۰۰)، انحراف معیار به شدت کاهش یافته و تخمین با دقت بسیار بالایی به شباهت حقیقی همگرا می‌گردد.
	\end{enumerate}
